**git-service.ts** — Git operations wrapper using `simple-git`:

```typescript
import simpleGit, { SimpleGit } from 'simple-git'
import { app } from 'electron'
import path from 'path'
import fs from 'fs/promises'

export class GitService {
  private baseDir: string

  constructor() {
    this.baseDir = path.join(app.getPath('userData'), 'codebase-analyzer', 'repos')
  }

  async ensureBaseDir(): Promise<void> {
    await fs.mkdir(this.baseDir, { recursive: true })
  }

  getRepoPath(repoName: string): string {
    return path.join(this.baseDir, repoName)
  }

  async clone(
    url: string,
    name: string,
    onProgress?: (data: { stage: string; progress: number; detail: string }) => void
  ): Promise<{ repoPath: string }> { ... }

  async getBranches(repoPath: string): Promise<{ name: string; current: boolean }[]> { ... }

  async checkout(repoPath: string, branch: string): Promise<void> { ... }

  async getCurrentCommit(repoPath: string): Promise<string> { ... }

  async getFileTree(repoPath: string): Promise<string[]> { ... }

  async getFileContent(repoPath: string, filePath: string): Promise<string> { ... }

  async removeRepo(repoPath: string): Promise<void> { ... }

  async fetchRemoteBranches(url: string): Promise<string[]> { ... }
}
```

Key implementation details:
- Clone with `['--depth', '100', '--single-branch']` by default
- Set env `GIT_TERMINAL_PROMPT=0` on every git instance to prevent hanging on auth prompts
- Use `git.outputHandler()` to parse stderr for progress and forward via `onProgress` callback
- `fetchRemoteBranches` uses `git ls-remote --heads <url>` to get branches BEFORE cloning (for the branch selection dropdown)
- `getFileTree` uses `git.raw(['ls-tree', '-r', '--name-only', 'HEAD'])` to list all tracked files
- `getFileContent` reads file with `fs.readFile(path.join(repoPath, filePath), 'utf-8')` with path traversal validation (resolved path must start with repoPath)
- `removeRepo` deletes the directory with `fs.rm(repoPath, { recursive: true, force: true })`

Detection strategy (check files from `fileList` and read key config files):
1. Check for `package.json` — if present, read it:
   - Has `react`, `next`, `vue`, `angular`, `svelte` in deps/devDeps → FE, framework = detected one
   - Has `express`, `fastify`, `nestjs`, `koa`, `hapi` → BE (Node), framework = detected one
2. Check for `pom.xml` — read and check for `spring-boot` → BE (Java), framework = 'spring-boot'
3. Check for `build.gradle` or `build.gradle.kts` — check for `spring` → BE (Java)
4. Check for `requirements.txt` or `pyproject.toml`:
   - Has `flask`, `django`, `fastapi` → BE (Python)
   - Has `airflow` → DE
5. Check for `go.mod` — read and check for web frameworks (gin, chi, echo, fiber) → BE (Go)
6. Check for `dags/` directory or `dbt_project.yml` → DE
7. Check for `App.tsx` or `App.jsx` → FE signal
8. If both FE and BE signals are HIGH → 'fullstack'
9. Return 'unknown' if no strong signals

Score each type independently (0-1). Highest score wins. If FE > 0.4 AND BE > 0.4 → fullstack.

**cache-db.ts** — SQLite cache for analysis results, following `NebulaDatabase` pattern:

```typescript
import Database from 'better-sqlite3'
import { app } from 'electron'
import path from 'path'

export class AnalyzerDatabase {
  private db: Database.Database

  constructor() {
    const dbPath = path.join(app.getPath('userData'), 'codebase-analyzer', 'analyzer.db')
    // Ensure parent directory exists
    const dir = path.dirname(dbPath)
    require('fs').mkdirSync(dir, { recursive: true })
    this.db = new Database(dbPath)
    this.db.pragma('journal_mode = WAL')
    this.db.pragma('foreign_keys = ON')
    this.initSchema()
  }

  private initSchema(): void { ... }
  getAnalysis(repoUrl: string, branch: string, commitSha: string): AnalysisResult | null { ... }
  saveAnalysis(repoUrl: string, branch: string, commitSha: string, result: AnalysisResult): void { ... }
  deleteAnalysis(repoUrl: string, branch?: string): void { ... }
  listCachedRepos(): { repoUrl: string; branch: string; commitSha: string; createdAt: string }[] { ... }

  // --- File indexing for FTS5 search (used by Plan 13-02 analyzer and Plan 13-05 Q&A) ---

  insertFileIndex(cacheId: string, filePath: string, content: string, language: string): void {
    this.db.prepare(
      'INSERT INTO file_index (cache_id, file_path, content, language) VALUES (?, ?, ?, ?)'
    ).run(cacheId, filePath, content, language)
  }

  clearFileIndex(cacheId: string): void {
    this.db.prepare('DELETE FROM file_index WHERE cache_id = ?').run(cacheId)
  }

  searchFiles(repoUrl: string, query: string): { filePath: string; snippet: string }[] {
    // Join file_index_fts with file_index and analysis_cache to filter by repoUrl
    const rows = this.db.prepare(`
      SELECT fi.file_path, snippet(file_index_fts, 1, '<mark>', '</mark>', '...', 64) AS snippet
      FROM file_index_fts
      JOIN file_index fi ON fi.rowid = file_index_fts.rowid
      JOIN analysis_cache ac ON ac.id = fi.cache_id
      WHERE ac.repo_url = ? AND file_index_fts MATCH ?
      LIMIT 20
    `).all(repoUrl, query) as { file_path: string; snippet: string }[]
    return rows.map(r => ({ filePath: r.file_path, snippet: r.snippet }))
  }

  close(): void { this.db.close() }
}
```

Schema tables:
- `analysis_cache`: `id TEXT PRIMARY KEY, repo_url TEXT, branch TEXT, commit_sha TEXT, result_json TEXT, created_at TEXT DEFAULT (datetime('now')), UNIQUE(repo_url, branch)`
- `file_index`: `id INTEGER PRIMARY KEY, cache_id TEXT REFERENCES analysis_cache(id) ON DELETE CASCADE, file_path TEXT, content TEXT, language TEXT` — for FTS5 search
- `file_index_fts`: FTS5 virtual table on file_index(file_path, content) for natural language code Q&A

ipcMain.handle('cban:analyze', async (event, repoPath: string, branch: string, repoUrl: string) => {
  const { analyzer } = getCbanInstances()
  const win = BrowserWindow.fromWebContents(event.sender)
  return analyzer.analyzeRepository(repoPath, branch, repoUrl, (progress) => {
    win?.webContents.send('cban:analysisProgress', progress)

// Get file content from a cloned repo
ipcMain.handle('cban:getFileContent', async (_event, repoPath: string, filePath: string) => {

  return { content, language, path: filePath, lineCount: content.split('\n').length }

  const { git } = getCbanInstances()

    repos: s.repos.filter((r) => r.id !== id),
    activeRepoId: s.activeRepoId === id ? null : s.activeRepoId,
    analysisResult: s.repos.find(r => r.id === id)?.id === s.activeRepoId ? null : s.analysisResult,
  })),
  setActiveRepo: (id) => set({ activeRepoId: id }),
  setAnalysisResult: (result) => set({ analysisResult: result }),
  setProgress: (progress) => set({ progress }),
  setError: (error) => set({ error }),
  setIsAnalyzing: (v) => set({ isAnalyzing: v }),

  openFile: (path, language) => set((s) => {
    const exists = s.openFiles.some((f) => f.path === path)
    return {
      openFiles: exists ? s.openFiles : [...s.openFiles, { path, language }],
      activeFilePath: path,
    }
  }),
  closeFile: (path) => set((s) => {
    const filtered = s.openFiles.filter((f) => f.path !== path)
    return {
      openFiles: filtered,
      activeFilePath: s.activeFilePath === path ? (filtered[filtered.length - 1]?.path ?? null) : s.activeFilePath,
    }
  }),
  setActiveFile: (path) => set({ activeFilePath: path }),
  setFileContent: (content) => set({ fileContent: content }),

  addQAMessage: (msg) => set((s) => ({ qaMessages: [...s.qaMessages, msg] })),
  updateLastQAMessage: (content) => set((s) => {
    const msgs = [...s.qaMessages]
    if (msgs.length > 0) msgs[msgs.length - 1] = { ...msgs[msgs.length - 1], content }
    return { qaMessages: msgs }
  }),
  clearQA: () => set({ qaMessages: [] }),
  setIsQAStreaming: (v) => set({ isQAStreaming: v }),
}))
