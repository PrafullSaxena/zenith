   ```ts
   type SettingsFieldType = 'text' | 'number' | 'boolean' | 'select' | 'password'
   interface SettingsField {
     key: string
     label: string
     type: SettingsFieldType
     description?: string
     defaultValue: unknown
     required?: boolean
     options?: { label: string; value: string }[]  // for 'select' type
     placeholder?: string
     validate?: (value: unknown) => string | null  // returns error message or null
   }
   ```

3. **PluginDefinition** — the core plugin registration type:
   ```ts
   interface PluginDefinition {
     id: PluginId
     name: string                          // Display name (e.g., "CodeReviewBot")
     description: string                   // Short description for tooltips
     icon: string                          // Lucide icon name (e.g., "GitPullRequest")
     route: string                         // React Router path (e.g., "/code-review-bot")
     component: React.LazyExoticComponent<React.ComponentType>  // Lazy-loaded view
     settingsSchema: SettingsField[]       // Plugin-specific settings fields
     defaultAgent: string | null           // Default AI agent provider id, null = user must pick
   }
   ```

Export all types. Use `import type { ComponentType, LazyExoticComponent } from 'react'` — do NOT import React as a value (tree-shaking).
