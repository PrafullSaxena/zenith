```tsx
import React, { useEffect, useRef, useState, useCallback } from 'react'
import './capture.css'

export default function CapturePopup(): React.JSX.Element {
  const [text, setText] = useState('')
  const [fromClipboard, setFromClipboard] = useState(false)
  const [submitting, setSubmitting] = useState(false)
  const textareaRef = useRef<HTMLTextAreaElement>(null)

  // Auto-focus and clipboard check on mount
  useEffect(() => {
    textareaRef.current?.focus()

    // Check clipboard once on open
    window.api.capture.getClipboard().then((clipText) => {
      if (clipText) {
        setText(clipText)
        setFromClipboard(true)
      }
    })
  }, [])

  // Auto-resize textarea as content grows
  useEffect(() => {
    const el = textareaRef.current
    if (!el) return
    el.style.height = 'auto'
    el.style.height = `${el.scrollHeight}px`
  }, [text])

  const handleClose = useCallback(() => {
    window.api.capture.close()
  }, [])

  const handleSubmit = useCallback(async () => {
    const trimmed = text.trim()
    if (!trimmed || submitting) return

    setSubmitting(true)
    try {
      await window.api.taskgroomer.createTask({
        text: trimmed,
        captureSource: fromClipboard && text === trimmed ? 'clipboard' : 'typed'
      })
    } finally {
      handleClose()
    }
  }, [text, fromClipboard, submitting, handleClose])

  const handleKeyDown = useCallback(
    (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
      if (e.key === 'Escape') {
        e.preventDefault()
        handleClose()
      } else if (e.key === 'Enter' && !e.shiftKey) {
        e.preventDefault()
        handleSubmit()
      }
      // Shift+Enter: default behavior (newline)
    },
    [handleClose, handleSubmit]
  )

  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    setText(e.target.value)
    // Badge disappears once user edits
    if (fromClipboard) setFromClipboard(false)
  }

  return (
    <div className="capture-overlay" onClick={handleClose}>
      <div
        className="capture-card"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="capture-header">
          <span className="capture-label">Capture task</span>
          {fromClipboard && (
            <span className="capture-badge">from clipboard</span>
          )}
        </div>
        <textarea
          ref={textareaRef}
          className="capture-input"
          value={text}
          onChange={handleChange}
          onKeyDown={handleKeyDown}
          placeholder="What needs doing?"
          rows={1}
          disabled={submitting}
        />
        <div className="capture-hint">
          <span>⏎ to capture</span>
          <span>esc to dismiss</span>
        </div>
      </div>
    </div>
  )
}
```

Key behaviors implemented per CONTEXT.md:
- Auto-focus on mount
- Clipboard auto-paste on mount (via capture:getClipboard IPC)
- "from clipboard" badge — disappears on first edit
- Enter → submit (no-op if empty)
- Shift+Enter → newline (default textarea behavior)
- Escape → close popup
- Auto-resize textarea (single-line expands to multi-line)
- captureSource: 'clipboard' when text matches original clipboard content, else 'typed'

In the JSX, add a route OUTSIDE the `<Route element={<AppLayout />}>` block. The route tree should look like:

```tsx
<HashRouter>
  <Routes>
    {/* Capture popup — no sidebar, no AppLayout */}
    <Route
      path="/capture"
      element={
        <Suspense fallback={null}>
          <ErrorBoundary>
            <CaptureApp />
          </ErrorBoundary>
        </Suspense>
      }
    />
    {/* Main app with layout */}
    <Route element={<AppLayout />}>
      {/* ... existing routes ... */}
    </Route>
  </Routes>
</HashRouter>
```

IMPORTANT: The /capture route must be added BEFORE the `<Route element={<AppLayout />}>` block so it takes precedence. The existing catch-all `path="*"` inside AppLayout will not intercept /capture since it's in a different Route group.
