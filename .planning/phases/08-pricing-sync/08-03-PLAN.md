```typescript
ipcMain.handle('launchpad:getSyncStatus', async () => {
  try {
    const providers: Array<'aws' | 'gcp' | 'azure'> = ['aws', 'gcp', 'azure']
    const statuses = providers.map(p => pricingRepository.getSyncStatus(p))
    return { success: true, statuses }
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err)
    return { success: false, error: message }
  }

// ── launchpad:getRegions ───────────────────────────────────────────────
ipcMain.handle('launchpad:getRegions', async (_event, provider: string) => {
  const validProviders = ['aws', 'gcp', 'azure']
  if (!validProviders.includes(provider)) {
    return { success: false, error: `Invalid provider: ${provider}` }
  }
  try {
    const regions = pricingRepository.getRegions(provider as 'aws' | 'gcp' | 'azure')
    return { success: true, regions }
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err)
    return { success: false, error: message }
  }
})
```

**Current launchpad section ends at `saveCredentials`. Add immediately after `saveCredentials`:**

```typescript
syncPricing: (): Promise<{
  success: boolean
  result?: {
    startedAt: number
    completedAt: number
    providers: Array<{
      provider: 'aws' | 'gcp' | 'azure'
      status: 'success' | 'error' | 'skipped'
      servicesUpdated: number
      error?: string
      deltaSkipped?: boolean
    }>
  }
  error?: string
}> =>
  ipcRenderer.invoke('launchpad:syncPricing'),

getSyncStatus: (): Promise<{
  success: boolean
  statuses?: Array<{
    provider: 'aws' | 'gcp' | 'azure'
    lastSyncAt: number | null
    status: string | null
    servicesUpdated: number | null
    error: string | null
  }>
  error?: string
}> =>
  ipcRenderer.invoke('launchpad:getSyncStatus'),

getRegions: (provider: string): Promise<{
  success: boolean
  regions?: Array<{ regionId: string; displayName: string }>
  error?: string
}> =>
  ipcRenderer.invoke('launchpad:getRegions', provider),

onSyncComplete: (
  callback: (result: {
    startedAt: number
    completedAt: number
    providers: Array<{ provider: string; status: string; servicesUpdated: number }>
  }) => void
): (() => void) => {
  const handler = (_event: Electron.IpcRendererEvent, result: unknown) =>
    callback(result as Parameters<typeof callback>[0])
  ipcRenderer.on('launchpad:syncComplete', handler)
  // Return an unsubscribe function so the renderer can clean up
  return () => ipcRenderer.removeListener('launchpad:syncComplete', handler)
},
```

**Placement:** Find the closing brace of the `launchpad` object in `preload/index.ts` (currently after `saveCredentials`). Insert the four new entries before the closing `}` of the `launchpad` block.

**Important — onSyncComplete pattern:** `ipcRenderer.on()` returns an `IpcRenderer` instance which `contextBridge` cannot proxy. The wrapper above returns a void-returning unsubscribe function instead — this follows the exact same pattern as other event listeners already established in the codebase (see the comment at the top of the preload file: "ipcRenderer.on() returns an IpcRenderer instance which contextBridge cannot proxy, so we must wrap those calls in void-returning functions").

**TypeScript note:** `Electron.IpcRendererEvent` may need to be imported. Check if `ipcRenderer` import at the top of the file already brings in the event type. If `Electron.IpcRendererEvent` causes a type error, use `import type { IpcRendererEvent } from 'electron'` at the top or cast the parameter as `unknown`.
