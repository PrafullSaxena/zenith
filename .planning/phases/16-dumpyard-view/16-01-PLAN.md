**Stale detection helper (pure function, exported):**
```typescript
export function isTaskStale(task: Task): boolean {
  // A dump task is stale if it has been in dump status for ≥3 days.
  // Use updatedAt if status was ever changed (user moved it back to dump);
  // otherwise createdAt. Since we don't track status-change timestamp separately,
  // use updatedAt as the staleness anchor (≥3 days in current status).
  // If task.status !== 'dump', never stale.
  if (task.status !== 'dump') return false
  const THREE_DAYS_MS = 3 * 24 * 60 * 60 * 1000
  return Date.now() - task.updatedAt >= THREE_DAYS_MS
}

export function staleDays(task: Task): number {
  return Math.floor((Date.now() - task.updatedAt) / (24 * 60 * 60 * 1000))
}
```

**Optimistic update pattern:**
```typescript
updateTaskStatus: async (id, status) => {
  const prev = get().tasks
  // Optimistic
  set({ tasks: prev.map(t => t.id === id ? { ...t, status, updatedAt: Date.now() } : t) })
  try {
    await window.api.taskgroomer.updateTask({ id, fields: { status } })
  } catch (err) {
    // Revert
    set({ tasks: prev, error: err instanceof Error ? err.message : 'Update failed' })
  }
}
```
