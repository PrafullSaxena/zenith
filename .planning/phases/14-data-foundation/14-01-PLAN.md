**Exports required:**

```ts
export interface Task {
  id: string
  text: string
  status: 'dump' | 'groomed' | 'done' | 'delegated' | 'aborted'
  captureSource: 'typed' | 'clipboard'
  createdAt: number   // Unix ms
  updatedAt: number   // Unix ms
  // Grooming metadata — nullable until Phase 18 AI agent writes them
  priority: 'p1' | 'p2' | 'p3' | null
  suggestedAction: 'do' | 'delegate' | 'defer' | 'delete' | null
  jiraTicketKey: string | null
  jiraTicketUrl: string | null
  evidenceSummary: string | null
  researchSummary: string | null
  researchLinks: string | null  // JSON array: {title, url}[]
  groomedAt: number | null      // Unix ms
}

export interface CreateTaskInput {
  text: string
  captureSource: 'typed' | 'clipboard'
}

export interface UpdateTaskInput {
  id: string
  fields: Partial<Omit<Task, 'id' | 'createdAt'>>
}
```

**Constructor:**
```ts
constructor() {
  const dbPath = path.join(app.getPath('userData'), 'tasks.db')
  this.db = new Database(dbPath)
  this.db.pragma('journal_mode = WAL')
  this.db.pragma('foreign_keys = ON')
  this.initSchema()
}
```

**Schema (initSchema):**
- Set `PRAGMA user_version = 1` after creation (same migration pattern as nebula)
- `CREATE TABLE IF NOT EXISTS tasks (...)` with all columns per CONTEXT.md decisions:
  - id TEXT PRIMARY KEY
  - text TEXT NOT NULL
  - status TEXT NOT NULL DEFAULT 'dump'
  - captureSource TEXT NOT NULL
  - createdAt INTEGER NOT NULL
  - updatedAt INTEGER NOT NULL
  - priority TEXT (nullable)
  - suggestedAction TEXT (nullable)
  - jiraTicketKey TEXT (nullable)
  - jiraTicketUrl TEXT (nullable)
  - evidenceSummary TEXT (nullable)
  - researchSummary TEXT (nullable)
  - researchLinks TEXT (nullable)
  - groomedAt INTEGER (nullable)
- `CREATE INDEX IF NOT EXISTS idx_tasks_status ON tasks(status)` — supports listTasks filter

**Methods:**

`createTask(input: CreateTaskInput): Task`
- Generate id with `crypto.randomUUID()` (built-in Node.js, no external dep)
- Set createdAt and updatedAt to `Date.now()`
- INSERT INTO tasks, return full Task object

`listTasks(statuses?: string[]): Task[]`
- If statuses provided and non-empty: `WHERE status IN (...)` using a prepared statement with parameter binding
- If no filter: return all rows
- No ORDER BY (renderer Zustand store handles sorting per CONTEXT.md decision)
- Map snake_case DB rows to camelCase Task interface

`updateTask(input: UpdateTaskInput): Task`
- Build SET clause dynamically from `input.fields` (exclude id, createdAt)
- Always set updatedAt = Date.now()
- Run UPDATE WHERE id = ?, then SELECT to return updated Task
- Throw if row not found (id doesn't exist)

`deleteTask(id: string): { success: boolean }`
- DELETE WHERE id = ?
- Return { success: true } (even if not found — idempotent delete is fine)

`close(): void`
- `this.db.close()`

**Column mapping note:** DB uses snake_case columns (capture_source, created_at, etc.). Map to camelCase in Task interface. Use a private `rowToTask(row: TaskRow): Task` helper.

Do NOT add FTS, graph edges, or any nebula-specific complexity. This is a simple CRUD table.

**Step 3 — Add four handlers inside `registerIpcHandlers()`, after the nebula handlers section. Add a `// --- Task Groomer channels ---` comment block:**

```ts
// --- Task Groomer channels ---
ipcMain.handle('taskgroomer:createTask', (_event, args: { text: string; captureSource: 'typed' | 'clipboard' }) => {
  return getTaskGroomerDb().createTask(args)
})

ipcMain.handle('taskgroomer:listTasks', (_event, args?: { statuses?: string[] }) => {
  return getTaskGroomerDb().listTasks(args?.statuses)
})

ipcMain.handle('taskgroomer:updateTask', (_event, args: { id: string; fields: Record<string, unknown> }) => {
  return getTaskGroomerDb().updateTask({ id: args.id, fields: args.fields as Parameters<typeof getTaskGroomerDb>['0']['fields'] })
})

ipcMain.handle('taskgroomer:deleteTask', (_event, args: { id: string }) => {
  return getTaskGroomerDb().deleteTask(args.id)
})
```

Use the actual UpdateTaskInput type from the import — adjust the type cast as needed to avoid `any`. The handlers are synchronous (better-sqlite3 is sync) so no `async` keyword needed.
