**PdfExporter (`pdf-exporter.ts`):**

This is a utility module, NOT a visual component.

```ts
interface PdfExportOptions {
  content: string  // markdown content
  metadata?: {
    title?: string
    author?: string
    date?: string
    plugin?: string
  }
  format?: 'report' | 'document' | 'diagram'  // default 'document'
}

export async function exportPdf(options: PdfExportOptions): Promise<void> {
  // Call the Electron main process via preload API
  // The actual PDF generation happens in main process
  await window.api.app.exportPdf({
    content: options.content,
    metadata: {
      title: options.metadata?.title ?? 'Zenith Export',
      author: options.metadata?.author ?? 'Zenith',
      date: options.metadata?.date ?? new Date().toISOString(),
      plugin: options.metadata?.plugin,
    },
    format: options.format ?? 'document',
  })
}
```
